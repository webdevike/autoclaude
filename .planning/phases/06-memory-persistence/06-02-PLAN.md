```typescript
/**
 * Load yesterday's and today's daily logs for session context.
 * Truncates each file to last 10KB to prevent context overflow.
 */
private loadRecentMemory(): string {
  const memoryDir = this.workspaceManager.getMemoryDir();

  const today = new Date();
  const yesterday = new Date(today);
  yesterday.setDate(yesterday.getDate() - 1);

  const todayFile = today.toISOString().split("T")[0] + ".md";
  const yesterdayFile = yesterday.toISOString().split("T")[0] + ".md";

  const MAX_BYTES = 10000; // 10KB per file

  let context = "";

  // Load yesterday's log
  const yesterdayPath = resolve(memoryDir, yesterdayFile);
  if (existsSync(yesterdayPath)) {
    let content = readFileSync(yesterdayPath, "utf-8");
    content = this.truncateToTail(content, MAX_BYTES);
    context += `\n## Yesterday's Notes (${yesterdayFile})\n\n${content}\n`;
  }

  // Load today's log
  const todayPath = resolve(memoryDir, todayFile);
  if (existsSync(todayPath)) {
    let content = readFileSync(todayPath, "utf-8");
    content = this.truncateToTail(content, MAX_BYTES);
    context += `\n## Today's Notes (${todayFile})\n\n${content}\n`;
  }

  return context;
}

/**
 * Truncate content to last N bytes (tail), preserving most recent entries.
 */
private truncateToTail(content: string, maxBytes: number): string {
  if (Buffer.byteLength(content) <= maxBytes) {
    return content;
  }

  const lines = content.split("\n").reverse();
  let truncated = "";
  let byteCount = 0;

  for (const line of lines) {
    const lineBytes = Buffer.byteLength(line + "\n");
    if (byteCount + lineBytes > maxBytes) break;
    truncated = line + "\n" + truncated;
    byteCount += lineBytes;
  }

  return `... (truncated: showing last ${Math.round(byteCount / 1024)}KB)\n\n${truncated}`;
}
```

```typescript
/**
 * File watcher for memory files — triggers reindexing when memory
 * files are created or modified. Phase 7 will provide the actual
 * indexer; this phase provides the hook infrastructure.
 */

import chokidar from "chokidar";
import { resolve } from "node:path";

export type MemoryFileChangedCallback = (filePath: string) => void;

/**
 * Watches MEMORY.md and memory/*.md for changes.
 * Calls onFileChanged callback with debouncing to prevent
 * duplicate triggers from atomic writes.
 */
export class MemoryWatcher {
  private watcher: chokidar.FSWatcher | null = null;
  private debounceTimer: ReturnType<typeof setTimeout> | null = null;

  constructor(
    private workspaceDir: string,
    private onFileChanged: MemoryFileChangedCallback
  ) {}

  /**
   * Start watching memory files.
   * Call BEFORE agent initialization to avoid missing early writes.
   */
  start(): void {
    const memoryDir = resolve(this.workspaceDir, "memory");
    const memoryMd = resolve(this.workspaceDir, "MEMORY.md");

    this.watcher = chokidar.watch([memoryMd, `${memoryDir}/*.md`], {
      persistent: true,
      ignoreInitial: true,
      ignored: ["**/*.tmp", "**/*.backup"],
      awaitWriteFinish: {
        stabilityThreshold: 500,
        pollInterval: 100,
      },
    });

    this.watcher.on("add", (path) => this.scheduleCallback(path));
    this.watcher.on("change", (path) => this.scheduleCallback(path));

    this.watcher.on("error", (err) => {
      console.warn(`[memory-watcher] Error: ${err}`);
    });

    console.log("[memory-watcher] Started watching memory files");
  }

  /**
   * Debounce callback — waits 1 second after last change before firing.
   * Prevents duplicate triggers from atomic write (temp + rename).
   */
  private scheduleCallback(filePath: string): void {
    if (this.debounceTimer) clearTimeout(this.debounceTimer);
    this.debounceTimer = setTimeout(() => {
      console.log(`[memory-watcher] File changed: ${filePath}`);
      try {
        this.onFileChanged(filePath);
      } catch (err) {
        console.warn(`[memory-watcher] Callback error: ${err}`);
      }
    }, 1000);
  }

  /**
   * Stop watching memory files. Call on shutdown.
   */
  stop(): void {
    if (this.debounceTimer) {
      clearTimeout(this.debounceTimer);
      this.debounceTimer = null;
    }
    if (this.watcher) {
      this.watcher.close();
      this.watcher = null;
      console.log("[memory-watcher] Stopped watching memory files");
    }
  }
}
```
