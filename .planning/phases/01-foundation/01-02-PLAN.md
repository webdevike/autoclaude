       - Add a `SessionManager` class (or module-level functions) in agent.ts:
         ```typescript
         import { appendFileSync, readFileSync, mkdirSync, existsSync } from 'node:fs';
         import { join } from 'node:path';
         import { homedir } from 'node:os';

         const SESSIONS_DIR = join(homedir(), '.jarvis', 'sessions');
         const MAX_MESSAGES = 50;

         function getSessionPath(userId: string): string {
           const userDir = join(SESSIONS_DIR, userId);
           if (!existsSync(userDir)) {
             mkdirSync(userDir, { recursive: true });
           }
           return join(userDir, 'messages.jsonl');
         }

         function appendSession(userId: string, entry: SessionEntry): void {
           const sessionPath = getSessionPath(userId);
           appendFileSync(sessionPath, JSON.stringify(entry) + '\n');
         }

         function loadSession(userId: string, limit = MAX_MESSAGES): SessionEntry[] {
           const sessionPath = getSessionPath(userId);
           try {
             const lines = readFileSync(sessionPath, 'utf8').split('\n').filter(Boolean);
             const entries: SessionEntry[] = [];
             // Take last `limit` lines
             for (const line of lines.slice(-limit)) {
               try {
                 entries.push(JSON.parse(line));
               } catch {
                 console.warn('[session] Skipping invalid JSONL entry');
               }
             }
             return entries;
           } catch (err) {
             if ((err as NodeJS.ErrnoException).code === 'ENOENT') {
               return [];
             }
             throw err;
           }
         }
         ```

         // In handleMessage(), add to the built-in command switch:
         case '/usage': {
           const arg = msg.text.split(' ')[1] || 'today'; // /usage today | /usage month
           const history = loadSession(msg.sender, Infinity); // load all entries
           const now = new Date();

           const entries = history.filter(e => {
             if (!e.usage) return false;
             const d = new Date(e.timestamp);
             if (arg === 'month') {
               return d.getMonth() === now.getMonth() && d.getFullYear() === now.getFullYear();
             }
             // default: today
             return d.toDateString() === now.toDateString();
           });

           // Aggregate per model
           const byModel: Record<string, { input: number; output: number; cost: number; requests: number }> = {};
           for (const e of entries) {
             const model = e.usage!.model;
             if (!byModel[model]) byModel[model] = { input: 0, output: 0, cost: 0, requests: 0 };
             byModel[model].input += e.usage!.inputTokens;
             byModel[model].output += e.usage!.outputTokens;
             byModel[model].cost += e.usage!.cost;
             byModel[model].requests += 1;
           }

           const totalCost = Object.values(byModel).reduce((sum, m) => sum + m.cost, 0);
           const period = arg === 'month' ? 'This month' : 'Today';

           let report = `${period}'s usage:\n`;
           for (const [model, stats] of Object.entries(byModel)) {
             report += `\n${model}: ${stats.requests} requests, ${stats.input} in / ${stats.output} out, $${stats.cost.toFixed(4)}`;
           }
           report += `\n\nTotal: $${totalCost.toFixed(4)}`;

           return { text: report };
         }
