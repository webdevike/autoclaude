    1. Update `packages/gateway/src/index.ts` `handleIncoming()` method to support streaming progress:

       - Change the `onProgress` callback type from `(status: string) => void` to accept `StreamProgressEvent` objects (import from @jarvis/core).

       - Implement a throttled edit strategy:
         ```typescript
         const EDIT_THROTTLE_MS = 1000; // Telegram rate limit: 1 edit/second
         let accumulated = "";
         let lastEditTime = 0;
         let editTimer: ReturnType<typeof setTimeout> | null = null;

         const onProgress = (event: StreamProgressEvent) => {
           if (event.type === 'text_delta' && event.delta) {
             accumulated += event.delta;
             const now = Date.now();

             if (now - lastEditTime >= EDIT_THROTTLE_MS) {
               // Enough time passed, edit immediately
               lastEditTime = now;
               channel.editMessage?.(message.sender, placeholderId!, accumulated).catch(() => {});
             } else if (!editTimer) {
               // Schedule an edit for when throttle period expires
               const remaining = EDIT_THROTTLE_MS - (now - lastEditTime);
               editTimer = setTimeout(() => {
                 lastEditTime = Date.now();
                 editTimer = null;
                 channel.editMessage?.(message.sender, placeholderId!, accumulated).catch(() => {});
               }, remaining);
             }
           } else if (event.type === 'tool_use' && event.toolName) {
             // Show tool usage status
             const statusText = accumulated
               ? `${accumulated}\n\n_Using tool: ${event.toolName}..._`
               : `Using tool: ${event.toolName}...`;
             channel.editMessage?.(message.sender, placeholderId!, statusText).catch(() => {});
             lastEditTime = Date.now();
           } else if (event.type === 'status' && event.text) {
             // Show status updates (e.g., "Triaging...", "Delegating...")
             channel.editMessage?.(message.sender, placeholderId!, event.text).catch(() => {});
             lastEditTime = Date.now();
           }
         };
         ```

       - After `orchestrator.handleMessage()` resolves, clear any pending edit timer and do a final edit with the complete response text.

       - Keep the existing fallback path: if no placeholder was created, just send the response as a new message.
